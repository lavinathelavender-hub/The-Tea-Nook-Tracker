function normalize(text) {
    return String(text).toLowerCase().replace(/[’']/g, "'").replace(/[^a-z0-9]+/g, " ").trim();
}

function showToast(text, type = "success") {
    const box = document.getElementById("toastContainer");
    const toast = document.createElement("div");
    toast.className = `toast ${type}`;
    toast.textContent = (type === "success" ? "✅ " : type === "error" ? "❌ " : "⚠️ ") + text;
    box.appendChild(toast);
    requestAnimationFrame(() => toast.classList.add("show"));
    setTimeout(() => {
        toast.classList.remove("show");
        setTimeout(() => toast.remove(), 250);
    }, 3500);
}

function msg(text, type = "success") {
    document.getElementById("adminMessage").textContent = text;
    showToast(text, type);
}

function databaseError(action, error) {
    console.error(action, error);
    msg(`${action}: ${error?.message || "Unknown database error"}`, "error");
}

function parseNotes(value) {
    if (Array.isArray(value)) return value.filter(Boolean);
    if (!value) return [];
    return String(value).split("|").map(x => x.trim()).filter(Boolean);
}

async function loadData(showFailure = true) {
    const { data: memberRows, error: memberError } = await supabaseClient
        .from("members")
        .select("id,name,notes")
        .order("id", { ascending: true });

    if (memberError) {
        if (showFailure) databaseError("Could not load members", memberError);
        return false;
    }

    const { data: flowerRows, error: flowerError } = await supabaseClient
        .from("flowers")
        .select("id,member_id,flower_name,points,bonus")
        .order("id", { ascending: true });

    if (flowerError) {
        if (showFailure) databaseError("Could not load flowers", flowerError);
        return false;
    }

    const grouped = new Map((memberRows || []).map(row => [row.id, {
        id: row.id,
        name: row.name,
        notes: parseNotes(row.notes),
        flowers: []
    }]));

    attributeBonuses.splice(0);
    for (const row of flowerRows || []) {
        const member = grouped.get(row.member_id);
        if (!member) continue;
        member.flowers.push([Number(row.points), row.flower_name, row.id]);
        if (Number(row.bonus) > 0) {
            attributeBonuses.push({
                member: member.name,
                flower: row.flower_name,
                bonus: Number(row.bonus),
                flowerId: row.id
            });
        }
    }

    members.splice(0, members.length, ...grouped.values());
    refreshAll();
    return true;
}

function refreshSelects() {
    const select = document.getElementById("memberSelect");
    const oldValue = select.value;
    select.innerHTML = "";
    [...members].sort((a, b) => a.name.localeCompare(b.name)).forEach(member => {
        select.add(new Option(member.name, String(member.id)));
    });
    if ([...select.options].some(option => option.value === oldValue)) select.value = oldValue;

    const datalist = document.getElementById("flowerSuggestions");
    datalist.innerHTML = "";
    const names = [...new Set(members.flatMap(member => member.flowers.map(flower => flower[1])))].sort((a, b) => a.localeCompare(b));
    names.forEach(name => {
        const option = document.createElement("option");
        option.value = name;
        datalist.appendChild(option);
    });
}

async function addMember() {
    const input = document.getElementById("newMember");
    const name = input.value.trim();
    if (!name) return msg("Enter a member name.", "error");
    if (members.some(member => normalize(member.name) === normalize(name))) return msg("A member with that name already exists.", "error");

    const { error } = await supabaseClient.from("members").insert({ name, notes: null });
    if (error) return databaseError("Could not add member", error);
    input.value = "";
    await loadData();
    msg(`Added ${name}.`);
}

async function renameMember(member, newName) {
    newName = newName.trim();
    if (!newName) return msg("Member name cannot be blank.", "error");
    if (members.some(item => item.id !== member.id && normalize(item.name) === normalize(newName))) return msg("A member with that name already exists.", "error");

    const oldName = member.name;
    const { error } = await supabaseClient.from("members").update({ name: newName }).eq("id", member.id);
    if (error) return databaseError("Could not rename member", error);
    await loadData();
    msg(`Member renamed from ${oldName} to ${newName}.`);
}

async function deleteMember(member) {
    if (!confirm(`Delete ${member.name} and all flowers?`)) return;

    const { error: flowerError } = await supabaseClient.from("flowers").delete().eq("member_id", member.id);
    if (flowerError) return databaseError("Could not delete the member's flowers", flowerError);

    const { error: memberError } = await supabaseClient.from("members").delete().eq("id", member.id);
    if (memberError) return databaseError("Could not delete member", memberError);

    await loadData();
    msg(`Member ${member.name} and all flower records deleted.`);
}

async function addFlower() {
    const memberId = Number(document.getElementById("memberSelect").value);
    const member = members.find(item => item.id === memberId);
    const nameInput = document.getElementById("newFlower");
    const flowerName = nameInput.value.trim();
    const points = Number(document.getElementById("newPoints").value);
    const bonus = Math.max(0, Number(document.getElementById("newBonus").value) || 0);

    if (!member || !flowerName) return msg("Choose a member and enter a flower.", "error");
    if (member.flowers.some(flower => normalize(flower[1]) === normalize(flowerName))) return msg("That member already has this flower.", "error");

    const { error } = await supabaseClient.from("flowers").insert({
        member_id: member.id,
        flower_name: flowerName,
        points,
        bonus
    });
    if (error) return databaseError("Could not add flower", error);

    nameInput.value = "";
    document.getElementById("newBonus").value = "0";
    await loadData();
    msg(`Added ${flowerName} to ${member.name}.`);
}

async function updateFlower(member, flower, row) {
    const controls = row.querySelectorAll("input,select");
    const flowerName = controls[0].value.trim();
    const points = Number(controls[1].value);
    const bonus = Math.max(0, Number(controls[2].value) || 0);
    const flowerId = flower[2];

    if (!flowerName) return msg("Flower name cannot be blank.", "error");
    if (member.flowers.some(item => item[2] !== flowerId && normalize(item[1]) === normalize(flowerName))) return msg("That member already has this flower.", "error");

    const { error } = await supabaseClient.from("flowers").update({
        flower_name: flowerName,
        points,
        bonus
    }).eq("id", flowerId);
    if (error) return databaseError("Could not update flower", error);

    await loadData();
    msg(`Flower ${flowerName} updated for ${member.name}.`);
}

async function deleteFlower(member, flower) {

}

async function deleteAllData() {
    if (!confirm("Delete ALL members and flowers from the shared Supabase database? This cannot be undone.")) return;

}

            const bonusInput = document.createElement("input");
            bonusInput.type = "number";
            bonusInput.min = "0";
            bonusInput.value = String(getAttributeBonus(member.name, flower[1]) || 0);
            const saveButton = document.createElement("button");
            saveButton.className = "admin-btn";
            saveButton.textContent = "Save";
            saveButton.onclick = () => updateFlower(member, flower, row);
            const removeButton = document.createElement("button");
            removeButton.className = "admin-btn danger";
            removeButton.textContent = "Delete";
            removeButton.onclick = () => deleteFlower(member, flower);

            row.append(flowerInput, pointsSelect, bonusInput, saveButton, removeButton);
            card.appendChild(row);
        });
        box.appendChild(card);
    });
}

function getAttributeBonus(memberName, flowerName) {
    const match = attributeBonuses.find(item => normalize(item.member) === normalize(memberName) && normalize(item.flower) === normalize(flowerName));
    return match ? match.bonus : null;
}

function getMemberBonusTotal(member) {
    return member.flowers.reduce((sum, [, flower]) => sum + (getAttributeBonus(member.name, flower) || 0), 0);
}

function getMemberAttributeCount(member) {
    return member.flowers.filter(([, flower]) => getAttributeBonus(member.name, flower)).length;
}

function getEmoji(name) {
    const lower = normalize(name);
    if (lower.includes("cherri")) return "🍒";
    if (lower.includes("peach")) return "🍑";
    if (lower.includes("snow")) return "❄️";
    if (lower.includes("bird")) return "🐦";
    if (lower.includes("lily")) return "🌼";
    if (lower.includes("rose")) return "🌹";
    if (lower.includes("aqua")) return "🌊";
    if (lower.includes("flower")) return "🌸";
    if (lower.includes("secret")) return "🌿";
    if (lower.includes("bree")) return "✨";
    if (lower.includes("yuki")) return "🫖";
    return "🌷";
}

function getFilteredFlowers(member) {
    let filteredFlowers = [...member.flowers];
    if (pointFilter.value !== "all") filteredFlowers = filteredFlowers.filter(([points]) => String(points) === pointFilter.value);
    if (attributeFilter.value === "hasAttributes") filteredFlowers = filteredFlowers.filter(([, flower]) => getAttributeBonus(member.name, flower));
    if (attributeFilter.value === "noAttributes") filteredFlowers = filteredFlowers.filter(([, flower]) => !getAttributeBonus(member.name, flower));
    const sortType = flowerSort.value;
    if (sortType === "pointsDesc") filteredFlowers.sort((a, b) => b[0] - a[0] || a[1].localeCompare(b[1]));
    if (sortType === "pointsAsc") filteredFlowers.sort((a, b) => a[0] - b[0] || a[1].localeCompare(b[1]));
    if (sortType === "nameAsc") filteredFlowers.sort((a, b) => a[1].localeCompare(b[1]) || b[0] - a[0]);
    if (sortType === "nameDesc") filteredFlowers.sort((a, b) => b[1].localeCompare(a[1]) || b[0] - a[0]);
    if (sortType === "bonusDesc") filteredFlowers.sort((a, b) => (getAttributeBonus(member.name, b[1]) || 0) - (getAttributeBonus(member.name, a[1]) || 0) || b[0] - a[0] || a[1].localeCompare(b[1]));
    return filteredFlowers;
}

function getSortedMembers(list) {
    const sorted = [...list];
    const sortType = memberSort.value;
    if (sortType === "nameAsc") sorted.sort((a, b) => a.name.localeCompare(b.name));
    if (sortType === "nameDesc") sorted.sort((a, b) => b.name.localeCompare(a.name));
    if (sortType === "countDesc") sorted.sort((a, b) => b.flowers.length - a.flowers.length || a.name.localeCompare(b.name));
    if (sortType === "countAsc") sorted.sort((a, b) => a.flowers.length - b.flowers.length || a.name.localeCompare(b.name));
    if (sortType === "bonusDesc") sorted.sort((a, b) => getMemberBonusTotal(b) - getMemberBonusTotal(a) || a.name.localeCompare(b.name));
    return sorted;
}

function renderMembers() {
    const search = normalize(searchInput.value);
    let filteredMembers = members.filter(member => {
        const flowerText = member.flowers.map(flower => flower.join(" ")).join(" ");
        const searchableText = normalize(member.name + " " + flowerText);
        const searchMatch = searchableText.includes(search);
        const pointMatch = pointFilter.value === "all" || member.flowers.some(([points]) => String(points) === pointFilter.value);
        const attributeMatch = attributeFilter.value === "all" ||
            (attributeFilter.value === "hasAttributes" && getMemberAttributeCount(member) > 0) ||
            (attributeFilter.value === "noAttributes" && member.flowers.some(([, flower]) => !getAttributeBonus(member.name, flower)));
        return searchMatch && pointMatch && attributeMatch;
    });
    filteredMembers = getSortedMembers(filteredMembers);
    grid.innerHTML = "";

    filteredMembers.forEach(member => {
        const shownFlowers = getFilteredFlowers(member);
        if (!shownFlowers.length) return;
        const card = document.createElement("article");
        card.className = "member-card";
        const attrCount = getMemberAttributeCount(member);
        const bonusTotal = getMemberBonusTotal(member);
        if (attrCount) card.classList.add("member-has-attributes");

        const header = document.createElement("div");
        header.className = "member-header";
        const title = document.createElement("h2");
        title.className = "member-name";
        title.textContent = `${getEmoji(member.name)} ${member.name}`;
        const count = document.createElement("span");
        count.className = "count-pill";
        count.textContent = `${shownFlowers.length} item${shownFlowers.length === 1 ? "" : "s"}`;
        header.append(title, count);
        card.appendChild(header);

        if (member.notes?.length) {
            const note = document.createElement("div");
            note.className = "attribute-summary";
            note.textContent = "📝 " + member.notes.join(" • ");
            card.appendChild(note);
        }
        if (attrCount) {
            const summary = document.createElement("div");
            summary.className = "attribute-summary";
            summary.textContent = `✨ ${attrCount} attribute flower${attrCount === 1 ? "" : "s"} | +${bonusTotal} total bonus`;
            card.appendChild(summary);
        }

        const list = document.createElement("ul");
        shownFlowers.forEach(([points, flower]) => {
            const bonus = getAttributeBonus(member.name, flower);
            const item = document.createElement("li");
            if (bonus) item.classList.add("has-attribute");
            const tag = document.createElement("span");
            tag.className = `tag r${points}`;
            tag.textContent = points;
            const flowerName = document.createElement("span");
            flowerName.textContent = flower;
            item.append(tag, flowerName);
            if (bonus) {
                const badge = document.createElement("span");
                badge.className = "attribute-badge";
                badge.textContent = `+${bonus}`;
                const label = document.createElement("span");
                label.className = "attribute-label";
                label.textContent = `${points}+${bonus} = ${points + bonus}`;
                item.append(badge, label);
            }
            list.appendChild(item);
        });
        card.appendChild(list);
        grid.appendChild(card);
    });

    emptyMessage.style.display = grid.children.length ? "none" : "block";
    updateStats();
}

function updateStats() {
    const allFlowers = members.flatMap(member => member.flowers.map(flower => ({ member: member.name, points: flower[0], flower: flower[1] })));
    const uniqueFlowers = new Set(allFlowers.map(item => normalize(item.flower)));
    const attributeMatches = allFlowers.filter(item => getAttributeBonus(item.member, item.flower));
    const highestCount = members.length ? Math.max(...members.map(member => member.flowers.length)) : 0;
    document.getElementById("memberCount").textContent = members.length;
    document.getElementById("flowerCount").textContent = allFlowers.length;
    document.getElementById("uniqueCount").textContent = uniqueFlowers.size;
    document.getElementById("attributeCount").textContent = attributeMatches.length;
    document.getElementById("topCount").textContent = highestCount;
}

function renderCommonFlowers() {
    const counts = {};
    members.forEach(member => member.flowers.forEach(([points, flower]) => {
        const key = normalize(flower);
        counts[key] = counts[key] || { flower, points, count: 0 };
        counts[key].count += 1;
    }));
    const common = Object.values(counts).sort((a, b) => b.count - a.count || b.points - a.points || a.flower.localeCompare(b.flower)).slice(0, 12);
    const container = document.getElementById("commonFlowers");
    container.innerHTML = "";
    common.forEach(item => {
        const chip = document.createElement("span");
        chip.className = "chip";
        chip.textContent = `${item.flower} - ${item.count}`;
        container.appendChild(chip);
    });
}

function renderOwnerLookup() {
    const query = normalize(ownerLookup.value);
    ownerResults.innerHTML = "";
    if (!query) {
        const hint = document.createElement("div");
        hint.className = "owner-row";
        hint.textContent = "Start typing a flower name to see matching members.";
        ownerResults.appendChild(hint);
        return;
    }
    const matches = [];
    members.forEach(member => member.flowers.forEach(([points, flower]) => {
        if (normalize(flower).includes(query)) matches.push({ member: member.name, flower, points, bonus: getAttributeBonus(member.name, flower) || 0 });
    }));
    if (!matches.length) {
        const none = document.createElement("div");
        none.className = "owner-row";
        none.textContent = "No matching flower owners found.";
        ownerResults.appendChild(none);
        return;
    }
    matches.sort((a, b) => b.bonus - a.bonus || b.points - a.points || a.member.localeCompare(b.member)).forEach(match => {
        const row = document.createElement("div");
        row.className = match.bonus ? "owner-row has-attribute-row" : "owner-row";
        row.textContent = match.bonus ? `${match.member} has ${match.points} - ${match.flower} (+${match.bonus}, total ${match.points + match.bonus})` : `${match.member} has ${match.points} - ${match.flower}`;
        ownerResults.appendChild(row);
    });
}
